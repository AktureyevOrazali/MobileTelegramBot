&НаКлиенте
Процедура ChatHTMLПриНажатии(Элемент, ДанныеСобытия, СтандартнаяОбработка)
    
    // 1. Перехват Enter (через v8action)
    Если ДанныеСобытия.Event <> Неопределено 
        И Найти(ДанныеСобытия.Event.url, "v8action://send-message") > 0 Тогда
        
        СтандартнаяОбработка = Ложь;
        ВыполнитьОтправкуИзHTML();
        Возврат;
	КонецЕсли;

    // 2. Перехват клика по кнопкам (Отправить, Завершить обращение)
    Если ДанныеСобытия.Anchor <> Неопределено Тогда
        Для Каждого Атрибут Из ДанныеСобытия.Anchor.attributes Цикл
            Если Атрибут.name = "data-button-id" И Атрибут.value = "Exchange" Тогда
                СтандартнаяОбработка = Ложь;
                ВыполнитьОтправкуИзHTML();
                Прервать;
            ИначеЕсли Атрибут.name = "data-button-id" И Атрибут.value = "CloseDialog" Тогда
                СтандартнаяОбработка = Ложь;
                ЗавершитьОбращение();
                Прервать;
            ИначеЕсли Атрибут.name = "data-button-id" И Атрибут.value = "AttachFile" Тогда
                СтандартнаяОбработка = Ложь;
                ВыбратьИЗагрузитьФайл();
                Прервать;
            ИначеЕсли Атрибут.name = "data-button-id" И Найти(Атрибут.value, "QuickCmd_") = 1 Тогда
                СтандартнаяОбработка = Ложь;
                НажатаБыстраяКоманда(Сред(Строка(Атрибут.value), 10));
                Прервать;
            ИначеЕсли Атрибут.name = "data-button-id" И Найти(Атрибут.value, "RemoveUpload_") = 1 Тогда
                СтандартнаяОбработка = Ложь;
                УдалитьОжидающееВложение(Число(Сред(Строка(Атрибут.value), 14)));
                Прервать;
            ИначеЕсли Атрибут.name = "data-button-id" И Найти(Атрибут.value, "Rate_") = 1 Тогда
                СтандартнаяОбработка = Ложь;
                НажатаОценка(Число(Сред(Строка(Атрибут.value), 6)));
                Прервать;
            ИначеЕсли Атрибут.name = "onclick" И СтрНайти(Атрибут.value, "openFullImage") > 0 Тогда
                СтандартнаяОбработка = Ложь;
                Попытка
                    // Извлекаем ID изображения из onclick атрибута
                    ТекстСобытия = Атрибут.value;
                    НачПоз = СтрНайти(ТекстСобытия, "openFullImage('") + 15;
                    КонПоз = СтрНайти(ТекстСобытия, "')");
                    Если НачПоз > 15 И КонПоз > НачПоз Тогда
                        IDКартинки = Сред(ТекстСобытия, НачПоз, КонПоз - НачПоз);
                        
                        // Получаем данные изображения через JavaScript
                        ДокументHTML = Элементы.ChatHTML.Документ;
                        Если ДокументHTML <> Неопределено Тогда
                            Попытка
                                СкрытыйЭлемент = ДокументHTML.getElementById(IDКартинки);
                                Если СкрытыйЭлемент <> Неопределено Тогда
                                    Ссылка = СкрытыйЭлемент.value;
                                    Base64 = СкрытыйЭлемент.getAttribute("data-base64");
                                    Mime = СкрытыйЭлемент.getAttribute("data-mime");
                                    ОткрытьИзображениеНаВесьЭкран(Ссылка, Base64, Mime);
                                КонецЕсли;
                            Исключение
                            КонецПопытки;
                        КонецЕсли;
                    КонецЕсли;
                Исключение
                КонецПопытки;
                Прервать;
            КонецЕсли;
        КонецЦикла;
    КонецЕсли;
КонецПроцедуры

&НаКлиенте
Процедура ВыполнитьОтправкуИзHTML()
    
    ДокументHTML = Элементы.ChatHTML.Документ;
    Если ДокументHTML = Неопределено Тогда
        Возврат;
    КонецЕсли;

    // Читаем текст
    Попытка
        Текст = ДокументHTML.forms["dataInput"].textInput.value;
    Исключение
        Текст = "";
    КонецПопытки;

    Текст = СокрЛП(Строка(Текст));
    Если ПустаяСтрока(Текст) И Не ЕстьОжидающиеВложенияНаСервере() Тогда
        Возврат;
    КонецЕсли;

    // Синхронизируем тему
    Попытка
        КлассBody = ДокументHTML.body.className;
        ЛокКласс = НРег(СокрЛП(КлассBody));
        Если Найти(ЛокКласс, "theme-light") > 0 Тогда
            Theme = "light";
        ИначеЕсли Найти(ЛокКласс, "theme-dark") > 0 Тогда
            Theme = "dark";
        КонецЕсли;
        ЗафиксироватьТемуНаСервере();
    Исключение
    КонецПопытки;

    // Отправляем
    MessageText = Текст;
    КомандаОтправить(Неопределено);

    // Очищаем поле ввода в HTML
    Попытка
        ДокументHTML.forms["dataInput"].textInput.value = "";
    Исключение
    КонецПопытки;

&НаКлиенте
Процедура ОткрытьИзображениеНаВесьЭкран(СсылкаНаИзображение, Base64Строка = "", MimeТип = "image/jpeg")
    Попытка
        Если Не ПустаяСтрока(Base64Строка) И Base64Строка <> "null" И Base64Строка <> "undefined" Тогда
            // Если есть base64 данные, создаем временный файл
            ДвоичныеДанные = Base64Значение(Base64Строка);
            
            // Определяем расширение по MimeType
            Расширение = ".jpg";
            Mime = НРег(MimeТип);
            
            Если СтрНайти(Mime, "png") > 0 Тогда
                Расширение = ".png";
            ИначеЕсли СтрНайти(Mime, "jpeg") > 0 Или СтрНайти(Mime, "jpg") > 0 Тогда
                Расширение = ".jpg";
            ИначеЕсли СтрНайти(Mime, "gif") > 0 Тогда
                Расширение = ".gif";
            ИначеЕсли СтрНайти(Mime, "bmp") > 0 Тогда
                Расширение = ".bmp";
            ИначеЕсли СтрНайти(Mime, "webp") > 0 Тогда
                Расширение = ".webp";
            Иначе
                Расширение = ".tmp";
            КонецЕсли;
            
            // Создаем временный файл с уникальным именем
            ВременныйКаталог = КаталогВременныхФайлов();
            ИмяФайла = ВременныйКаталог + "img_" + Строка(Новый УникальныйИдентификатор()) + Расширение;
            
            ДвоичныеДанные.Записать(ИмяФайла);
            
            // Открываем файл в ассоциированной программе
            ЗапуститьПриложение(ИмяФайла);
            
            // Добавляем файл в список для удаления
            ДобавитьФайлВСписокДляУдаления(ИмяФайла);
            
            // Запускаем обработчик для удаления файлов через 30 секунд
            ПодключитьОбработчикОжидания("УдалитьВременныеФайлы", 30);
            
        ИначеЕсли Не ПустаяСтрока(СсылкаНаИзображение) И СсылкаНаИзображение <> "null" И СсылкаНаИзображение <> "undefined" Тогда
            // Если есть ссылка, открываем её в браузере
            ЗапуститьПриложение(СсылкаНаИзображение);
        КонецЕсли;
    Исключение
        Сообщить("Не удалось открыть изображение: " + ОписаниеОшибки());
    КонецПопытки;

&НаКлиенте
Процедура УдалитьВременныеФайлы() Экспорт
    Попытка
        // Получаем список файлов для удаления
        СписокФайлов = ПолучитьСписокФайловДляУдаления();
        
        Если СписокФайлов.Количество() = 0 Тогда
            ОтключитьОбработчикОжидания("УдалитьВременныеФайлы");
            Возврат;
        КонецЕсли;
        
        НовыйСписок = Новый Массив;
        ФайлыУдалены = 0;
        
        // Пытаемся удалить каждый файл
        Для Каждого ИмяФайла Из СписокФайлов Цикл
            Попытка
                Если ФайлСуществует(ИмяФайла) Тогда
                    УдалитьФайлы(ИмяФайла);
                    ФайлыУдалены = ФайлыУдалены + 1;
                КонецЕсли;
            Исключение
                // Файл еще занят, оставляем в списке
                НовыйСписок.Добавить(ИмяФайла);
            КонецПопытки;
        КонецЦикла;
        
        // Сохраняем оставшиеся файлы
        СохранитьСписокФайловДляУдаления(НовыйСписок);
        
        // Если остались файлы, подключаем обработчик снова
        Если НовыйСписок.Количество() > 0 Тогда
            ПодключитьОбработчикОжидания("УдалитьВременныеФайлы", 30);
        Иначе
            ОтключитьОбработчикОжидания("УдалитьВременныеФайлы");
        КонецЕсли;
        
    Исключение

Функция ОпределитьРеальноеРасширение(ДвоичныеДанные)

    СтрокаБайт = ПолучитьБайтыКакСтроку(ДвоичныеДанные);
    
    // JPEG: первые 2 байта 0xFF 0xD8
    Если Сред(СтрокаБайт, 1, 2) = "FFD8" Тогда
        Возврат ".jpg";
    КонецЕсли;
    
    // PNG: первые 4 байта 0x89 0x50 0x4E 0x47
    Если Сред(СтрокаБайт, 1, 4) = "89504E47" Тогда
        Возврат ".png";
    КонецЕсли;
    
    // GIF: первые 3 байта "GIF"
    Если Сред(СтрокаБайт, 1, 3) = "474946" Тогда
        Возврат ".gif";
    КонецЕсли;
    
    // BMP: первые 2 байта 0x42 0x4D
    Если Сред(СтрокаБайт, 1, 2) = "424D" Тогда
        Возврат ".bmp";
    КонецЕсли;
    
    Возврат ".tmp";

    Ширина = 320;
    Высота = 240;

    Если ЗначениеЗаполнено(ШиринаИсходная) И ЗначениеЗаполнено(ВысотаИсходная)

function insertEmoji(emoji) {
  var input = document.getElementById('messageInput');

    input.value = text.substring(0, start) + emoji + text.substring(end);
    input.selectionStart = input.selectionEnd = start + emoji.length;
  } else {
    input.value = input.value + emoji;

  input.focus();
  closeEmojiPanel();

/* отправка по Enter - совместимо со старым IE и русской клавиатурой */
function handleKeyPress(e) {
  var event = e || window.event;
  var keyCode = event.keyCode || event.which || 0;
  
  // Enter = keyCode 13 (работает для любой раскладки)
  if (keyCode === 13) {
    // Предотвращаем стандартное поведение формы
    if (event.preventDefault) {
      event.preventDefault();
    }
    if (event.stopPropagation) {
      event.stopPropagation();
    }
    event.returnValue = false;
    event.cancelBubble = true;

    var input = document.getElementById('messageInput');
    if (!input) return false;
    
    // trim для старого IE
    var val = input.value;
    if (val) {
      val = val.replace(/^\s+|\s+$/g, '');
    }
    // Кликаем по ссылке "Отправить" - 1С перехватит это событие
    var sendLink = document.getElementById('sendLink');
    if (sendLink) {
      // Для старого IE используем fireEvent или click()
      if (sendLink.click) {
        sendLink.click();
      } else if (document.createEvent) {
        var evt = document.createEvent('MouseEvents');
        evt.initMouseEvent('click', true, true, window, 0, 0, 0, 0, 0, false, false, false, false, 0, null);
        sendLink.dispatchEvent(evt);
      } else if (sendLink.fireEvent) {
        sendLink.fireEvent('onclick');
      }
    }
    return false;
  }
  return true;
}

/* анимация появления элементов - IE7-совместимо (filter:alpha) */
function fadeInMessages() {
  try {
    var chat = document.getElementById('chatMessages');
    if (!chat) return;
    var items = chat.getElementsByTagName('div');
    var targets = [];
    for (var i = 0; i < items.length; i++) {
      if (items[i].className && items[i].className.indexOf('msg-new') >= 0) {
        targets.push(items[i]);
      }
    }
    if (targets.length === 0) return;

    // Начальное состояние: невидимо
    for (var j = 0; j < targets.length; j++) {
      targets[j].style.visibility = 'visible';
      if (targets[j].style.filter !== undefined) {
        targets[j].style.filter = 'alpha(opacity=0)';
      }
      if (targets[j].style.opacity !== undefined) {
        targets[j].style.opacity = '0';
      }
    }

    // Поэтапный fade-in
    var s = 0;
    var opacityInterval = window.setInterval(function() {
      s += 25;
      for (var k = 0; k < targets.length; k++) {
        var el = targets[k];
        if (el.style.filter !== undefined) {
          el.style.filter = 'alpha(opacity=' + s + ')';
        }
        if (el.style.opacity !== undefined) {
          el.style.opacity = (s / 100);
        }
        if (s >= 100) {
          el.className = el.className.replace(/\s*msg-new\s*/g, ' ');
        }
      }
      if (s >= 100) {
        window.clearInterval(opacityInterval);
      }
    }, 70);
  } catch (e) {}
}

/**
 * Обработчик paste на поле ввода.
 * Пробует обнаружить изображение в данных clipboard.
 * Если изображение не найдено — не мешает стандартной вставке текста.
 */
function handlePaste(e) {
    try {
        var clipData = (e && e.clipboardData) ? e.clipboardData : null;
        if (!clipData) return; // Нет данных — пусть работает стандартно

        // Способ 1: Современный API (clipboardData.items)
        if (clipData.items && clipData.items.length > 0) {
            for (var i = 0; i < clipData.items.length; i++) {
                if (clipData.items[i].type && clipData.items[i].type.indexOf('image') !== -1) {
                    if (e.preventDefault) e.preventDefault();
                    e.returnValue = false;

                    var file = clipData.items[i].getAsFile();
                    if (!file) continue;

                    var reader = new FileReader();
                    reader.onload = function(ev) {
                        _sendPastedImage(ev.target.result);
                    };
                    reader.readAsDataURL(file);
                    return;
                }
            }
        }

        // Способ 2: Проверяем наличие текста в буфере
        var hasText = false;
        try {
            var txt1 = clipData.getData('Text');
            var txt2 = clipData.getData('text/plain');
            if ((txt1 && txt1.length > 0) || (txt2 && txt2.length > 0)) {
                hasText = true;
            }
        } catch (ex) {}

        // Если это не текст и современные API не нашли картинку - 
        // используем запасной механизм IE через скрытый div
        if (!hasText) {
            if (e.preventDefault) e.preventDefault();
            e.returnValue = false;
            
            // Запускаем асинхронно, чтобы избежать зависания MSHTML при смене фокуса
            window.setTimeout(function() {
                triggerIEPaste();
            }, 10);
        }

    } catch (err) {}
}

    triggerIEPaste();
}

/**
 * Вставляет изображение через hidden contenteditable div (fallback для старых IE/MSHTML).
 * Выполняется асинхронно, чтобы не "повесить" движок 1С.
 */
function triggerIEPaste() {
    var pasteDiv = document.getElementById('pasteTarget');
    var input = document.getElementById('messageInput');
    if (!pasteDiv) return;

    try {
        pasteDiv.innerHTML = '';
        pasteDiv.focus();
        
        // Программно вызываем вставку (в IE сработает для картинок)
        document.execCommand('paste');

        // Ждем пока IE отрендерит <img>
        window.setTimeout(function() {
            try {
                var imgs = pasteDiv.getElementsByTagName('img');
                for (var i = 0; i < imgs.length; i++) {
                    var src = imgs[i].src || imgs[i].getAttribute('src') || '';
                    // IE вставляет картинки в виде base64 данных
                    if (src.indexOf('data:image') === 0) {
                        _sendPastedImage(src);
                        break;
                    }
                }
            } catch (ex) {}

            pasteDiv.innerHTML = '';
            if (input) input.focus();
        }, 100);
    } catch (e) {
        if (input) input.focus();
    }
}

/* Извлекает base64 и mime из data-URL и отправляет в 1С */
function _sendPastedImage(dataUrl) {
    try {
        if (!dataUrl || dataUrl.indexOf('data:') !== 0) return;
        var commaPos = dataUrl.indexOf(',');
        if (commaPos < 0) return;
        var meta   = dataUrl.substring(0, commaPos);
        var base64 = dataUrl.substring(commaPos + 1);
        if (!base64 || base64.length < 10) return;

        var mimeType = 'image/png';
        var mm = meta.match(/data:([^;,]+)/);
        if (mm && mm[1]) mimeType = mm[1];

        var ext = (mimeType.split('/')[1] || 'png');
        if (ext === 'jpeg') ext = 'jpg';
        var fileName = 'pasted_' + new Date().getTime() + '.' + ext;

        if (window.external && window.external.Call) {
            window.external.Call("ЗагрузитьИзображениеИзБуфера", fileName, base64, mimeType);
        }
    } catch (ex) {}
}

/* Настройка обработчика вставки — только на поле ввода, без перехвата keydown */
function setupPasteHandler() {
    var input = document.getElementById('messageInput');
    if (input) {
        if (input.addEventListener) {
            input.addEventListener('paste', handlePaste, false);
        } else if (input.attachEvent) {
            input.attachEvent('onpaste', handlePaste);
        }
    }
}

                window.external.Call("ОткрытьИзображениеНаВесьЭкран", imageUrl, "", "");

        }
